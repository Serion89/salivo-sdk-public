Part I: Language Fundamentals & Core Syntax

1.1 Lexical Structure & Clean Language Design

Salivo enforces a clean, modern type system while providing maximum flexibility for functions, variables, and system programming:

  • Data Types (Zero-Underscore Standard): Structs, enums, traits, and type annotations strictly use clean PascalCase without underscores (Point, Option, Result, UserProfile). The underscore _ is strictly never a datatype.
  • Functions, Variables, & Struct Fields (Flexible Naming): Functions, method names, variables, and struct fields freely support both modern camelCase and snake_case with underscores (pathJoin or path_join, assertEq or assert_eq, userId or user_id, maxLineWidth or max_line_width). Both forms are fully first-class and interoperable across the standard library, tooling, and user code.
  • Numbers & Wildcards: Numbers support underscore digit separators for readability (e.g., 1_000_000), and _ is used for pattern match wildcards and discarded bindings (let _ = expr;).
salivo 
// ✅ Clean Data Types (Zero-Underscores)
struct UserSession {
    int userId;
    string token;
    bool isActive;
}

// ✅ Flexible Function & Variable Identifiers
pub func get_user_session(int user_id) -> UserSession {
    let timeout_ms = 30_000;
    // ...
}

Comments

salivo

/*
  Multi-line block comment
*/

/// Documentation docstring attached to following declaration
pub func calculate() -> int { return 42; }

1.2 Modules & Stream Import Directives

Every Salivo file begins with an explicit module namespace declaration. Modules are organized in dot-separated hierarchies.

Module Declaration

salivo 
module myapp.network.client;

Stream Import Syntax (><)

Salivo uses the Stream Header Import Operator (><) exclusively for importing modules, types, and functions. (The legacy import keyword is deprecated).

salivo 
// 1. Full module stream import
><salivo.std.io;
><salivo.std.math;

// 2. Selective symbol import with destructuring
><salivo.std.core::{assert, assertEq, Option, some, none};

// 3. Selective cryptographic and filesystem primitives
><salivo.std.crypto::{sha256Hex, secSeed};
><salivo.std.fs::{pathJoin, pathExists, open, close};

// 4. Custom package imports
><mypackage.services.auth::{authenticate, Session};

1.3 Variables, Mutability, & Scope

Variables in Salivo are immutable by default when declared with let.

salivo 
// Immutable binding (cannot be reassigned)
let max_connections = 100;
let host = "127.0.0.1";

// Mutable binding
let mut counter = 0;
counter = counter + 1;
counter += 5;

// Explicit type annotations
int port = 8080;
float ratio = 1.618;
bool is_running = true;
string application_name = "SalivoEngine";

// Compile-time constants
const BUFFER_CAPACITY = 8192;
const MAX_RETRIES = 3;

// Global package-scoped bindings
global let SYSTEM_VERSION = "1.0.0";
global int ACTIVE_WORKERS = 0;

1.4 Data Types & Literals

Primitive Data Types

  • int: 64-bit signed two's complement integer (-9223372036854775808 to 9223372036854775807).
  • float: 32-bit IEEE 754 single-precision floating point.
  • double: 64-bit IEEE 754 double-precision floating point.
  • bool: Boolean values true or false.
  • char: Unicode scalar codepoint value (represented internally as UTF-32 scalar).
  • string: UTF-8 immutable reference-counted string sequence.
  • byte: 8-bit unsigned octet (0 to 255).
  • void: Unit type representing the absence of a value.

Literals

salivo

let flag = true;

1.7 Functions & Procedures

Functions are declared with func. The return type is defined after ->. If omitted, the return type defaults to void.

salivo

func calculate_area(width: int, height: int) -> int {
    return width * height;
}

// Public generic function
pub func identity<T>(item: T) -> T {
    return item;
}

// Function with no return value
pub func log_message(msg: string) {
    outln($"[LOG] {msg}");
}

// Program Entry Point
func main() -> int {
    let area = calculate_area(10, 20);
    outln($"Area: {area}");
    return 0;
}

2.3 Pattern Matching (match)

Pattern matching is exhaustive and supports destructuring enum variants and constants.

salivo 
pub func handle_response(resp: WebResponse) -> string {
    return match resp {
        WebResponse::Success(body) => $"200 OK: {body}",
        WebResponse::Redirect(url, code) => $"Redirect {code} to {url}",
        WebResponse::ClientError(status, err) => $"Error {status}: {err}",
        WebResponse::ServerError(msg) => $"500 Internal Error: {msg}",
        _ => "Unknown status",
    };
}

let status = HttpMethod::Get;
match status {
    HttpMethod::Get => outln("Performing GET request"),
    HttpMethod::Post => outln("Performing POST request"),
    _ => outln("Other HTTP method"),
}

2.4 Traits & Method Implementations

Traits define contracts for polymorphism. Methods and traits are attached to structs using impl.

salivo 
pub trait Printable {
    func tostring(self) -> string;
}

pub struct User {
    id: int,
    username: string,
}

// Implementing a Trait
impl Printable for User {
    func tostring(self) -> string {
        return $"User(id={self.id}, username={self.username})";
    }
}

// Adding Methods Directly to a Struct
impl User {
    func new(id: int, name: string) -> User {
        return User { id: id, username: name };
    }

    func isadmin(self) -> bool {
        return self.id == 1;
    }
}

// Usage
let u = User::new(1, "guest");
outln(u.tostring());
if (u.isadmin()) {
    outln("Administrator access granted.");
}

// Automatic Scope Cleanup Protocol (Drop Trait)
pub trait Drop {
    func drop(self);
}

impl Drop for User {
    func drop(self) {
        outln($"Cleaning up resources for user {self.username}...");
    }
}

// Display and Debug Traits for Formatting
pub trait Display {
    func display(self) -> String;
}

pub trait Debug {
    func debug(self) -> String;
}

impl Display for User {
    func display(self) -> String {
        return $"User({self.username} #{self.id})";
    }
}

impl Debug for User {
    func debug(self) -> String {
        return $"User {{ id: {self.id}, username: \"{self.username}\" }}";
    }
}

3.3 Formatted String Interpolation ($"...")

Templates embed variables, expressions, and modifier chains directly:

salivo 
let user = "  developer  ";
let score = 98.6;
let rank = 1;

let summary = $"Candidate: {user:trim:upper}, Rank: #{rank:str}, Score: {score:float}";
outln(summary);
// Output: Candidate: DEVELOPER, Rank: #1, Score: 98.6

3.4 String Literal Auto-Coercion

Salivo features automatic string literal coercion. Whenever a function, struct field, or variable declaration expects a borrowed StrSlice or owned String, a raw "string literal" is seamlessly coerced at compile time without requiring manual conversion boilerplate:

salivo 
><salivo.std.fs::{pathJoin};
><salivo.std.strings::{pushStr, stringNew};

// String literals automatically coerce to StrSlice / String parameters!
let joined = pathJoin("var", "log/salivo.log");

let mut s = stringNew();
s = pushStr(s, "Automated Coercion");

3.5 The format! Macro & Trait Formatting

Salivo provides a high-performance, compile-time validated format! macro that expands expressions into owned String instances with zero unnecessary allocations.

Placeholders & Dispatch Modes

  • {} — Display formatting: Invokes the Display trait (func display(self) -> String) on custom types, or native string formatting on primitives (int, float, bool, string, StrSlice, String).
  • {:?} — Debug formatting: Invokes the Debug trait (func debug(self) -> String) on custom types, or escaped quote debugging on strings.
salivo 
><salivo.std.core::{Display, Debug, outln};
><salivo.std.strings::{String};

pub struct Point {
    int x;
    int y;
}

impl Display for Point {
    func display(self) -> String {
        return $"Point({self.x}, {self.y})";
    }
}

pub func main() -> int {
    let p = Point { x: 10, y: 25 };
    let text = format!("Coordinate: {}, Label: {:?}", p, "target_alpha");
    outln(text);
    // Output: Coordinate: Point(10, 25), Label: "target_alpha"
    return 0;
}

Safety & Memory Invariants

1. Checked Allocation Arithmetic: Aggregate segment lengths and final buffer allocations are evaluated through runtime-checked arithmetic (salivo_checked_add_len), preventing 64-bit integer wrapping vulnerabilities and panicking cleanly if sizes exceed memory limits.

2. Deterministic Temporary Deallocation: Temporary heap allocations produced dynamically during formatting (such as custom Display returned String buffers or escaped debug C buffers) are tracked and freed exactly once upon formatting completion.

3. Authoritative Length with Embedded NUL: Formatting preserves exact byte counts and embedded NUL (\0) bytes without strlen truncation.

Part IV: Low-Level Memory Engine & Smart Pointers

The salivo.std.mem module gives full control over heap memory, reference counting, and raw layout intrinsics.

4.1 Heap Allocation Engine

salivo 
><salivo.std.mem;

// Allocate 64 bytes with 8-byte alignment
let addr = alloc(64, 8);

// Allocate zero-initialized memory
let zero_addr = alloczero(128, 8);

// Check if pointer is currently live
let is_live = isallocated(addr);

// Reallocate buffer to larger size (from 64 to 256 bytes)
let new_addr = realloc(addr, 64, 256, 8);

// Deallocate when finished
dealloc(new_addr, 256, 8);
dealloc(zero_addr, 128, 8);

4.2 Raw Memory Operations & Intrinsics

salivo 
><salivo.std.mem;

let buffer = alloc(32, 8);

// Write and read 64-bit integers at offsets
writei64(buffer, 0, 1000);
writei64(buffer, 8, 2000);
let val0 = readi64(buffer, 0); // 1000
let val1 = readi64(buffer, 8); // 2000

// Write and read single bytes
writebyte(buffer, 16, 255);
let b = readbyte(buffer, 16);  // 255

// Zero out a region of memory
zero(buffer, 0, 32);

// Copy memory from source to destination
let dest = alloc(32, 8);
copy(buffer, dest, 32);

// Pointer arithmetic and string conversions
let s_ptr = strtoptr("Hello Memory");
let str_back = ptrtostr(s_ptr);

dealloc(buffer, 32, 8);
dealloc(dest, 32, 8);

4.3 Slices & Spans

Non-owning views over contiguous memory buffers:

salivo 
><salivo.std.mem;

let raw = alloc(80, 8);
for (let mut i = 0; i < 10; i = i + 1) {
    writei64(raw, i * 8, (i + 1) * 10);
}

// Create slice view
let s: Slice<int> = fromrawparts<int>(raw, 10);
let length = slicelen<int>(s);      // 10
let el0 = sliceget<int>(s, 0);       // 10
let el4 = sliceget<int>(s, 4);       // 50

// Modify through slice view
sliceset<int>(s, 0, 999);

// Create subslice (indices 2 to 6)
let sub = slicesub<int>(s, 2, 6);

dealloc(raw, 80, 8);

4.4 Smart Pointers: Box<T>, Arc<T>, Weak<T>, Arena

salivo 
><salivo.std.mem;

// 1. Box<T> (Heap allocated single ownership)
let b = boxnew<int>(42, 8);
let value = deref<int>(b); // 42

// 2. Arc<T> (Atomic Reference Counting)
let arc_node = arcnew<string>("Shared State", 16);
let cloned_arc = retain<string>(arc_node);
let strong_cnt = strongcount<string>(cloned_arc); // 2

let released = release<string>(cloned_arc);
let final_release = release<string>(arc_node); // Automatically deallocated

// 3. Arena Allocator
let a = arena(1048576); // 1 MB Arena

Every module in the Salivo Standard Library (salivo.std.*) is described below with complete function signatures, descriptions, and code examples.

5.1 salivo.std.prelude & salivo.std.core

Automatically imported into every Salivo compilation unit.

Functions, Structs & Traits

Option<T>

salivo 
pub struct Option<T> {
    value: T,
    is_some: bool,
}
  • some<T>(val: T) -> Option<T>: Constructs an Option containing a value.
  • none<T>() -> Option<T>: Constructs an empty Option.

Result<T, E>

salivo 
pub struct Result<T, E> {
    value: T,
    error: E,
    is_ok: bool,
}
  • ok<T, E>(val: T) -> Result<T, E>: Constructs a successful Result.
  • err<T, E>(e: E) -> Result<T, E>: Constructs a failed Result.

Diagnostics & Assertions

  • panic(msg: string): Aborts execution with a diagnostic trace.
  • assert(cond: bool, msg: string): Panics if cond is false.
  • assertEq(a: int, b: int, msg: string) / asserteq(...): Panics if a != b.
  • assertStrEq(a: string, b: string, msg: string) / assertstreq(...): Panics if a != b.
  • assertNe(a: int, b: int, msg: string) / assertne(...): Panics if a == b.
  • assertStrNe(a: string, b: string, msg: string) / assertstrne(...): Panics if a == b.

Standard Traits

  • trait Display { func display(self) -> string; }
  • trait Debug { func debug(self) -> string; }
  • trait Iterator<T> { func next(self) -> Option<T>; }
  • trait Drop { func drop(self); }: Standard automated RAII cleanup protocol.

Core Conversions & Formatting

  • str(v: int) -> string: Converts integer to string.
  • intstr(v: int) -> string: Converts integer to string.
  • strint(s: string) -> int: Parses string to integer.
  • floatstr(f: float) -> string: Converts float to string.
  • strfloat(s: string) -> float: Parses string to float.
  • len(s: string) -> int: Returns string length.
  • trim(s: string) -> string: Strips whitespace from both ends.
  • upper(s: string) -> string: Converts string to uppercase.
  • lower(s: string) -> string: Converts string to lowercase.
  • slice(s: string, start: int, end: int) -> string: Substring from start to end.
  • charat(s: string, idx: int) -> int: Returns Unicode codepoint at index idx.
  • find(s: string, sub: string) -> int: Returns first index of sub, or -1.
  • contains(s: string, sub: string) -> bool: Returns true if sub exists in s.
  • startswith(s: string, prefix: string) -> bool / startsWith(...): Returns true if s begins with prefix.
  • endswith(s: string, suffix: string) -> bool / endsWith(...): Returns true if s terminates with suffix.
  • replace(s: string, target: string, repl: string) -> string: Replaces occurrences of target with repl.
  • repeat(s: string, count: int) -> string: Repeats string count times.
  • padstart(s: string, targetlen: int, pad: string) -> string / padStart(...): Prepends pad until targetlen.
  • padend(s: string, targetlen: int, pad: string) -> string / padEnd(...): Appends pad until targetlen.

NOTE
Unicode Case Mapping Architecture & Structural Scope:
unicodeToUpper, unicodeToLower, toUpper, and toLower provide high-performance Unicode-aware 1:1 scalar code point case mapping across major global scripts (ASCII, Latin-1 Supplement, Latin Extended-A, Greek/Coptic, Cyrillic, and Fullwidth Latin).
Because scalar functions map a single code point to a single code point (int -> int), 1:1 scalar conversion preserves code point length invariants. Multi-code point expansions (such as German ß expanding to SS, or ligatures such as ﬁ expanding to FI) and contextual/locale-specific casing (such as Turkish dotted/dotless I/i) are outside the 1:1 scalar mapping model. Unmapped code points outside the curated set are safely preserved without modification or corruption.
TIP
Authoritative Length & Embedded NUL (\0) Preservation:
Both String ({ ptr: int, len: int, cap: int }) and StrSlice ({ ptr: int, len: int }) treat their explicit len field as authoritative across all compiler lowering passes and standard library operations. Unlike legacy C strings, strings in Salivo fully support arbitrary binary byte sequences and embedded NUL bytes (\0) without premature strlen() truncation.

6.2 Compiler Optimization Pipeline & Performance Tuning

The Salivo compiler infrastructure leverages an advanced Static Single Assignment (SSA) optimization engine feeding into LLVM 16:

  • Optimization Levels (-O 0 to -O 3):
  • -O 0 (Debug): Fast compilation, zero instruction reordering, preserves full debug line tables.
  • -O 1 (Basic): Local peephole optimizations and dead value stripping.
  • -O 2 (Standard Release): Global common subexpression elimination, function inlining, and standard loop simplifications.
  • -O 3 / --opt-level aggressive: Aggressive loop unrolling, SLP auto-vectorization, memory-to-register promotion, and aggressive inlining.
  • Host CPU Vectorization (--native): Emits code specifically tailored to the host instruction set architecture, leveraging AVX2, FMA, and BMI2 vector registers.
  • Zero-Cost SSA Optimization Passes:
  • ArcOptimization: Analyzes lifetime ranges and elides redundant paired ARC retain/release operations within basic blocks and loops, reducing atomic reference counting overhead to zero for local variables.
  • BoundsCheckElimination: Employs dominator tree range analysis to verify index validity at loop entry, safely hoisting bounds checks out of inner iteration loops.
  • ConstantDivisorOptimization: Evaluates constant non-zero divisors at compile time, eliminating branch conditions and panic splitting on arithmetic division and modulo operations.
  • DeadCodeElimination (DCE): Recursively strips dead branches, unreachable blocks, and unused temporary values prior to backend emission.

Example 1: High-Performance Data Pipeline with Collections & Hashing

salivo 
module app.pipeline;

><salivo.std.core::{assert, assertEq};
><salivo.std.collections::{vecNew, push, get, veclen, mapNew, put, lookup};
><salivo.std.crypto::{sha256Hex};
><salivo.std.time::{now, elapsed};

func main() -> int {
    let start_time = now();
    outln("=== Starting Salivo Data Pipeline ===");

    // Initialize collections
    let mut data_vec = vecNew<string>();
    data_vec = push<string>(data_vec, "transaction_1001");
    data_vec = push<string>(data_vec, "transaction_1002");
    data_vec = push<string>(data_vec, "transaction_1003");

    let mut ledger = mapNew<string, string>();

    // Process and hash each item
    for (let mut i = 0; i < veclen<string>(data_vec); i = i + 1) {
        let raw_tx = get<string>(data_vec, i);
        let digest = sha256Hex(raw_tx);
        ledger = put<string, string>(ledger, raw_tx, digest);
        outln($"[PROCESSED] {raw_tx:upper} -> Digest: {digest:slice(0, 16)}...");
    }

    // Verify lookup
    let query = lookup<string, string>(ledger, "transaction_1002");
    if (query.is_some) {
        outln($"Verified lookup for TX 1002: {query.value:slice(0, 8)}");
    }

    let dur = elapsed(start_time);
    outln($"Pipeline completed in {dur.nanoseconds} ns");
    return 0;
}

Example 2: Thread-Safe State & Atomic Concurrency

salivo 
module app.concurrent_counter;

><salivo.std.core::{assertEq};
><salivo.std.sync::{mutex, lock, unlock, atomic, fetchadd, load};
><salivo.std.task::{scope, spawntask, scopespawn, scopejoin, Task};

func main() -> int {
    outln("=== Salivo Concurrency & Sync Demonstration ===");

    let total_tasks = 10;
    let global_counter = atomic(0);
    let state_lock = mutex();

    let sc = scope();

    for (let mut i = 0; i < total_tasks; i = i + 1) {
        // Increment atomic counter safely
        fetchadd(global_counter, 1);
        
        let guard = lock(state_lock);
        outln($"Task {i:str} registered. Counter value: {load(global_counter):str}");
        unlock(guard);
    }

    let final_count = load(global_counter);
    assertEq(final_count, total_tasks, "Concurrency counter mismatch");
    outln($"All {final_count:str} operations completed successfully.");
    return 0;
}
